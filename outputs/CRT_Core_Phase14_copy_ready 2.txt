// ─────────────────────────────────────────────────────────────────────────────
// CRT Indicator — Phases 1-14
//
// Phase 1 (CRT core):
//   - CRT reference range (last confirmed source candle)
//   - CRT high / CRT low; sweep above/below; reclaim (close back inside)
//   - Configurable CRT settings + clean visualization
//
// Phase 2 (Liquidity):
//   - Swing highs/lows (configurable pivot sensitivity, non-repainting)
//   - Equal highs/lows (configurable ATR tolerance)
//   - Previous Day High/Low (PDH/PDL)
//   - Buy-Side (BSL, above) / Sell-Side (SSL, below) liquidity levels
//   - Sweep detection, with optional close-back-through confirmation
//   - Show/hide toggles + clutter caps
//
// Phase 3 (Market Structure):
//   - Separate structural pivots (independent sensitivity from liquidity)
//   - HH/HL/LH/LL swing classification
//   - BOS (Break of Structure) / CHoCH (Change of Character) detection
//   - Persistent structure bias (bullish / bearish / neutral)
//   - Non-repainting: structural pivots only actionable after confirmation
//
// Phase 4 (Liquidity Engine):
//   - Separate liquidity pivots with own sensitivity
//   - BSL/SSL level tracking with active/swept/consumed states
//   - Equal highs/lows (tick-based tolerance)
//   - Sweep = wick grab + close rejection; consumed = close through
//   - Non-repainting: levels only active after pivot confirmation
//
// Phase 5 (Setup Quality Engine):
//   - Confluence scoring (liquidity + structure + bias + displacement + session)
//   - Reversal vs continuation classification
//   - Anti-chase filter, cooldown, duplicate prevention
//   - Alert conditions for qualified and strong setups
//
// Phase 6 (Trade Engine):
//   - Hypothetical trade simulation from Phase 5 setups
//   - ATR or structure stop, R-multiple target, max holding period
//   - Win rate, expectancy, profit factor, max drawdown (R)
//   - Performance by direction, score bucket, and setup type
//
// Phase 14 (Historical Validation):
//   - Explicit inclusive date-range comparison of Control/BOS/Retest/Confirm
//   - Exchange-timezone boundaries and FULL/PARTIAL chart-data coverage audit
//   - Measurement only; no changes to Phase 1-13 signals or trade decisions
//
// Deliberately NOT included yet (future phases):
//   position sizing, compounding, FVG, order blocks, optimization.
//
// Design rules honored:
//   - indicator() only, never strategy(); no orders / broker execution.
//   - No future data: HTF values are read from the PREVIOUS closed HTF bar via
//     the `expr[1]` + lookahead_on idiom (never reads the developing HTF bar).
//   - Minimize repaint: sweep/reclaim evaluated on confirmed bars by default.
//   - Historical events stay fixed: per-range state latches and only the active
//     box is mutated; older boxes are frozen once a new range starts.
// ─────────────────────────────────────────────────────────────────────────────

// ── Inputs: CRT range ────────────────────────────────────────────────────────
string GRP_CRT   = "CRT range"
string crtTfInput     = input.timeframe("", "CRT timeframe", group = GRP_CRT, tooltip = "Blank = chart timeframe (2-candle CRT). Set a higher timeframe (e.g. 60, 240, D) to use the last closed HTF candle as the range. Lower-than-chart timeframes are ignored.")

bool   wickSweep      = input.bool(true, "Sweep uses wick", group = GRP_CRT, tooltip = "On: the wick (high/low) must pierce the CRT boundary. Off: the bar must CLOSE beyond the boundary.")
int    maxRanges      = input.int(15, "Max CRT ranges kept", minval = 1, maxval = 100, group = GRP_CRT, tooltip = "How many recent CRT boxes stay on the chart (older ones are removed to respect object limits).")

bool  showLevels          = input.bool(true, "Show CRT high/low lines", group = GRP_VIS)
bool  showSweepMarkers    = input.bool(true, "Show sweep markers", group = GRP_VIS)
bool  showReclaimMarkers  = input.bool(true, "Show reclaim markers", group = GRP_VIS)
color rangeColor          = input.color(#4c8bf5, "Range color", group = GRP_VIS)
color sweepColor          = input.color(#ff9800, "Sweep color", group = GRP_VIS)
color reclaimColor        = input.color(#26a69a, "Reclaim color", group = GRP_VIS)
int   boxTransp           = input.int(88, "Box fill transparency", minval = 0, maxval = 100, group = GRP_VIS)

// ── Resolve CRT source timeframe (ignore anything below the chart TF) ─────────
string crtTfRaw = crtTfInput == "" ? timeframe.period : crtTfInput
bool   crtTfTooLow = timeframe.in_seconds(crtTfRaw) < timeframe.in_seconds()
string crtTf = crtTfTooLow ? timeframe.period : crtTfRaw

// Previous CLOSED source candle -> the CRT reference range.
// `expr[1]` + lookahead_on returns the last closed HTF bar with no lag and no
// future leak (we never touch the currently developing HTF bar).
[srcHigh, srcLow, srcTime] = request.security(syminfo.tickerid, crtTf, [high[1], low[1], time[1]], lookahead = barmerge.lookahead_on)

// New source candle detected (safe every-bar comparison, no ta.* in conditionals).
bool isNewSrc = not na(srcTime) and (na(srcTime[1]) or srcTime != srcTime[1])

// Process live bar or only confirmed bars, per setting.
bool processBar = confirmOnClose ? barstate.isconfirmed : true

// ── Active CRT range state (latched per range) ───────────────────────────────
var float crtHigh       = na
var float crtLow        = na
var int   crtStartTime  = na
var bool  sweptHigh     = false
var bool  sweptLow      = false
var bool  reclaimedHigh = false
var bool  reclaimedLow  = false
var box   activeBox     = na
var array<box> crtBoxes = array.new_box()

// True when the bar closes back within [crtLow, crtHigh].
isInside(float c, float hi, float lo) =>
    not na(c) and not na(hi) and not na(lo) and c <= hi and c >= lo

// Probe used to detect a sweep: wick (high/low) or close, per setting.
float probeHigh = wickSweep ? high : close
float probeLow  = wickSweep ? low  : close

// Per-bar events (reset each bar so plotshape marks only the triggering bar).
bool sweepHighEvent   = false
bool sweepLowEvent    = false
bool reclaimHighEvent = false
bool reclaimLowEvent  = false

if processBar and not na(srcHigh) and not na(srcLow) and not na(srcTime)
    // Establish / roll the CRT range on a new source candle.
    if na(crtHigh) or isNewSrc
        crtHigh       := srcHigh
        crtLow        := srcLow
        crtStartTime  := srcTime
        sweptHigh     := false
        sweptLow      := false
        reclaimedHigh := false
        reclaimedLow  := false
        if showBox
            activeBox := box.new(crtStartTime, crtHigh, time, crtLow, xloc = xloc.bar_time, border_color = rangeColor, bgcolor = color.new(rangeColor, boxTransp))
            array.push(crtBoxes, activeBox)
            if array.size(crtBoxes) > maxRanges
                box.delete(array.shift(crtBoxes))

    // Detect sweep, then reclaim, against the active range.
    if not na(crtHigh) and not na(crtLow)
        sweepHighEvent := probeHigh > crtHigh and not sweptHigh
        sweepLowEvent  := probeLow  < crtLow  and not sweptLow
        if sweepHighEvent
            sweptHigh := true
        if sweepLowEvent
            sweptLow := true

        bool inside = isInside(close, crtHigh, crtLow)
        reclaimHighEvent := sweptHigh and inside and not reclaimedHigh
        reclaimLowEvent  := sweptLow  and inside and not reclaimedLow
        if reclaimHighEvent
            reclaimedHigh := true
        if reclaimLowEvent
            reclaimedLow := true

        // Only the active box is mutated; older boxes stay frozen.
        if showBox and not na(activeBox)
            color stateColor = (reclaimedHigh or reclaimedLow) ? reclaimColor : (sweptHigh or sweptLow) ? sweepColor : rangeColor
            box.set_right(activeBox, time)
            box.set_border_color(activeBox, stateColor)
            box.set_bgcolor(activeBox, color.new(stateColor, boxTransp))

// ── Levels: step lines that hold each range's high/low while it is active ─────
plot(showLevels ? crtHigh : na, "CRT high", color = rangeColor, style = plot.style_stepline, linewidth = 1)
plot(showLevels ? crtLow  : na, "CRT low",  color = rangeColor, style = plot.style_stepline, linewidth = 1)

// ── Event markers (events, not trade signals) ────────────────────────────────
plotshape(showSweepMarkers and sweepHighEvent, title = "Sweep above CRT high", style = shape.circle, location = location.abovebar, color = sweepColor, size = size.tiny)
plotshape(showSweepMarkers and sweepLowEvent,  title = "Sweep below CRT low",  style = shape.circle, location = location.belowbar, color = sweepColor, size = size.tiny)
plotshape(showReclaimMarkers and reclaimHighEvent, title = "Reclaim after high sweep", style = shape.diamond, location = location.abovebar, color = reclaimColor, size = size.tiny)
plotshape(showReclaimMarkers and reclaimLowEvent,  title = "Reclaim after low sweep",  style = shape.diamond, location = location.belowbar, color = reclaimColor, size = size.tiny)

// PHASE 2 — LIQUIDITY
// Swing/equal highs & lows, previous-day high/low, BSL/SSL, and sweep detection.
// Non-repainting: pivots confirm `pivotRight` bars after they form, and sweeps
// are evaluated on the same confirmed-bar gate (`processBar`) used by Phase 1.
// ═════════════════════════════════════════════════════════════════════════════

// ── Inputs: liquidity ────────────────────────────────────────────────────────
string GRP_LIQ = "Liquidity"
bool  showSwings      = input.bool(true,  "Show swing highs/lows", group = GRP_LIQ)
bool  showEqual       = input.bool(true,  "Show equal highs/lows", group = GRP_LIQ, tooltip = "Highlights consecutive swings that sit at (nearly) the same price - a stronger liquidity pool.")
bool  showPDHL        = input.bool(true,  "Show previous day high/low", group = GRP_LIQ)
bool  showLiqSweeps   = input.bool(true,  "Show liquidity sweep markers", group = GRP_LIQ)
bool  showSweptLevels = input.bool(true,  "Keep swept levels (faded)", group = GRP_LIQ, tooltip = "On: a swept level freezes at the sweep bar and fades. Off: it is removed on sweep.")
bool  showLiqLabels   = input.bool(false, "Show PDH/PDL labels", group = GRP_LIQ)
int   pivotLeft       = input.int(5, "Pivot left bars", minval = 1, group = GRP_LIQ, tooltip = "Bars to the LEFT required for a swing. Higher = fewer, stronger swings.")
int   pivotRight      = input.int(5, "Pivot right bars", minval = 1, group = GRP_LIQ, tooltip = "Bars to the RIGHT required to CONFIRM a swing. A swing is only drawn this many bars later (non-repainting).")
float eqTolAtr        = input.float(0.10, "Equal tolerance (x ATR)", minval = 0.0, step = 0.05, group = GRP_LIQ, tooltip = "Two consecutive swings are 'equal' when within this multiple of ATR.")
int   eqAtrLen        = input.int(14, "Equal tolerance ATR length", minval = 1, group = GRP_LIQ)
bool  confirmLiqSweep = input.bool(false, "Confirm sweep (close back through)", group = GRP_LIQ, tooltip = "Off: sweep when the wick pierces the level. On: sweep only once a bar CLOSES back through the level (grab + rejection).")
int   maxLiqPerSide   = input.int(6,  "Max active levels per side", minval = 1, maxval = 50, group = GRP_LIQ)
int   maxSweptKept    = input.int(10, "Max swept levels kept", minval = 0, maxval = 100, group = GRP_LIQ)
color bslColor        = input.color(#089981, "BSL color (buy-side, above)", group = GRP_LIQ)
color sslColor        = input.color(#f23645, "SSL color (sell-side, below)", group = GRP_LIQ)
color eqColor         = input.color(#ab47bc, "Equal-level color", group = GRP_LIQ)
color sweptLiqColor   = input.color(#787b86, "Swept-level color", group = GRP_LIQ)

// ── Liquidity model ──────────────────────────────────────────────────────────
// A resting liquidity level. isBuy = above price (BSL) vs below price (SSL).
type LiqLevel
    float price
    int   startTime
    int   confirmBar
    bool  isBuy
    bool  isEqual
    bool  pierced
    line  ln

var array<LiqLevel> bslActive = array.new<LiqLevel>()
var array<LiqLevel> sslActive = array.new<LiqLevel>()

// Most recent confirmed swing on each side (used for equal-level tests).
var float lastSwingHigh     = na
var int   lastSwingHighTime = na
var float lastSwingLow      = na
var int   lastSwingLowTime  = na

// Diagnostic storage: keep active liquidity levels in arrays.
// The visual segment starts on the confirmation bar and remains visible historically.
addLevel(array<LiqLevel> arr, float price, int startTime, int confirmBar, bool isBuy, bool isEqual, line ln) =>
    array.push(arr, LiqLevel.new(price, startTime, confirmBar, isBuy, isEqual, false, ln))

// Stop a diagnostic line exactly on the sweep bar.
// During this test build, swept historical lines remain visible and are NOT faded/deleted.
retireLine(line ln, int x2Val) =>
    if not na(ln)
        line.set_extend(ln, extend.none)
        line.set_x2(ln, x2Val)

// ── Pivots + equal-level test (computed every bar, acted on confirmed bars) ──
float pivotHi = ta.pivothigh(pivotLeft, pivotRight)
float pivotLo = ta.pivotlow(pivotLeft, pivotRight)
float atrEq   = ta.atr(eqAtrLen)

// Previous-day high/low from the last CLOSED daily bar (non-repainting idiom).
[pdHigh, pdLow] = request.security(syminfo.tickerid, "D", [high[1], low[1]], lookahead = barmerge.lookahead_on)
bool isNewDay  = timeframe.change("D")
bool pdEnabled = timeframe.in_seconds() <= timeframe.in_seconds("D")

// Persistent PDH/PDL state.
var float pdHighPrice   = na
var float pdLowPrice    = na
var line  pdHighLine    = na
var line  pdLowLine     = na
var label pdHighLabel   = na
var label pdLowLabel    = na
var bool  pdHighPierced = false
var bool  pdLowPierced  = false
var bool  pdHighSwept   = false
var bool  pdLowSwept    = false

// Per-bar liquidity sweep events (for markers; reused by later phases).
bool bslSweepEvent = false
bool sslSweepEvent = false

// Per-bar pivot confirmation events (for diagnostic markers).
bool pivotHiEvent = false
bool pivotLoEvent = false

if processBar
    // New swing high -> BSL level (+ equal-high test vs the prior swing high).
    if not na(pivotHi)
        pivotHiEvent := true
        int  hiTime   = time[pivotRight]
        bool isEqHigh = not na(lastSwingHigh) and not na(atrEq) and math.abs(pivotHi - lastSwingHigh) <= eqTolAtr * atrEq
        bool showHigh = dmLiqVis and (showSwings or (isEqHigh and showEqual))
        line hiLine = na
        if showHigh
            color hiCol = (isEqHigh and showEqual) ? eqColor : bslColor
            // The pivot itself occurred `pivotRight` bars ago, but it only becomes
            // actionable liquidity NOW, on this confirmation bar.
            hiLine := line.new(
                 x1 = bar_index,
                 y1 = pivotHi,
                 x2 = bar_index + 1,
                 y2 = pivotHi,
                 xloc = xloc.bar_index,
                 extend = extend.right,
                 color = hiCol,
                 width = 2,
                 style = line.style_solid)
        addLevel(bslActive, pivotHi, hiTime, bar_index, true, isEqHigh, hiLine)
        lastSwingHigh     := pivotHi
        lastSwingHighTime := hiTime

    // New swing low -> SSL level (+ equal-low test vs the prior swing low).
    if not na(pivotLo)
        pivotLoEvent := true
        int  loTime  = time[pivotRight]
        bool isEqLow = not na(lastSwingLow) and not na(atrEq) and math.abs(pivotLo - lastSwingLow) <= eqTolAtr * atrEq
        bool showLow = dmLiqVis and (showSwings or (isEqLow and showEqual))
        line loLine = na
        if showLow
            color loCol = (isEqLow and showEqual) ? eqColor : sslColor
            // Same rule as BSL: the historical pivot marker is back-plotted,
            // but the liquidity level only becomes actionable on confirmation.
            loLine := line.new(
                 x1 = bar_index,
                 y1 = pivotLo,
                 x2 = bar_index + 1,
                 y2 = pivotLo,
                 xloc = xloc.bar_index,
                 extend = extend.right,
                 color = loCol,
                 width = 2,
                 style = line.style_solid)
        addLevel(sslActive, pivotLo, loTime, bar_index, false, isEqLow, loLine)
        lastSwingLow     := pivotLo
        lastSwingLowTime := loTime

    // BSL sweeps: wick above the level; optional confirm = close back below.
    if array.size(bslActive) > 0
        for i = array.size(bslActive) - 1 to 0
            LiqLevel lvl = array.get(bslActive, i)
            if not na(lvl.ln)
                line.set_x2(lvl.ln, bar_index)
            // Never let the confirmation bar sweep the level it just created.
            bool eligible = bar_index > lvl.confirmBar
            if eligible and high > lvl.price
                lvl.pierced := true
            bool swept = eligible and (confirmLiqSweep ? (lvl.pierced and close < lvl.price) : (high > lvl.price))
            if swept
                bslSweepEvent := true
                retireLine(lvl.ln, bar_index)
                array.remove(bslActive, i)

    // SSL sweeps: wick below the level; optional confirm = close back above.
    if array.size(sslActive) > 0
        for i = array.size(sslActive) - 1 to 0
            LiqLevel lvl = array.get(sslActive, i)
            if not na(lvl.ln)
                line.set_x2(lvl.ln, bar_index)
            // Same protection for sell-side liquidity.
            bool eligible = bar_index > lvl.confirmBar
            if eligible and low < lvl.price
                lvl.pierced := true
            bool swept = eligible and (confirmLiqSweep ? (lvl.pierced and close > lvl.price) : (low < lvl.price))
            if swept
                sslSweepEvent := true
                retireLine(lvl.ln, bar_index)
                array.remove(sslActive, i)

    // Previous-day high/low: always detect when TF is suitable; only draw when visible.
    if pdEnabled and isNewDay and not na(pdHigh) and not na(pdLow)
        if not na(pdHighLine)
            line.delete(pdHighLine)
        if not na(pdLowLine)
            line.delete(pdLowLine)
        if not na(pdHighLabel)
            label.delete(pdHighLabel)
        if not na(pdLowLabel)
            label.delete(pdLowLabel)
        pdHighLine  := na
        pdLowLine   := na
        pdHighLabel := na
        pdLowLabel  := na
        pdHighPrice   := pdHigh
        pdLowPrice    := pdLow
        pdHighPierced := false
        pdLowPierced  := false
        pdHighSwept   := false
        pdLowSwept    := false
        if showPDHL
            pdHighLine := line.new(time, pdHigh, time, pdHigh, xloc = xloc.bar_time, extend = extend.right, color = bslColor, width = 1, style = line.style_dotted)
            pdLowLine  := line.new(time, pdLow,  time, pdLow,  xloc = xloc.bar_time, extend = extend.right, color = sslColor, width = 1, style = line.style_dotted)
            if showLiqLabels
                pdHighLabel := label.new(time, pdHigh, "PDH", xloc = xloc.bar_time, style = label.style_none, textcolor = bslColor, size = size.tiny)
                pdLowLabel  := label.new(time, pdLow,  "PDL", xloc = xloc.bar_time, style = label.style_none, textcolor = sslColor, size = size.tiny)

    if pdEnabled and not na(pdHighPrice) and not pdHighSwept
        if not na(pdHighLine)
            line.set_x2(pdHighLine, time)
        if high > pdHighPrice
            pdHighPierced := true
        bool sweptPdh = confirmLiqSweep ? (pdHighPierced and close < pdHighPrice) : (high > pdHighPrice)
        if sweptPdh
            pdHighSwept   := true
            bslSweepEvent := true
            retireLine(pdHighLine, time)
            pdHighLine := na

    if pdEnabled and not na(pdLowPrice) and not pdLowSwept
        if not na(pdLowLine)
            line.set_x2(pdLowLine, time)
        if low < pdLowPrice
            pdLowPierced := true
        bool sweptPdl = confirmLiqSweep ? (pdLowPierced and close > pdLowPrice) : (low < pdLowPrice)
        if sweptPdl
            pdLowSwept    := true
            sslSweepEvent := true
            retireLine(pdLowLine, time)
            pdLowLine := na

// ── Liquidity sweep markers (events, not trade signals) ──────────────────────
plotshape(dmLiqVis and showLiqSweeps and bslSweepEvent, title = "BSL swept", style = shape.xcross, location = location.abovebar, color = bslColor, size = size.tiny)
plotshape(dmLiqVis and showLiqSweeps and sslSweepEvent, title = "SSL swept", style = shape.xcross, location = location.belowbar, color = sslColor, size = size.tiny)

// ============================================================================
// PHASE 3 — MARKET STRUCTURE
// Confirmed structural pivots, HH/HL/LH/LL classification, BOS/CHoCH detection.
// Uses a SEPARATE set of pivots from Phase 2 liquidity (independent sensitivity).
// Non-repainting: structural pivots only become usable after ta.pivothigh/
// ta.pivotlow confirms them (structureRight bars later). BOS/CHoCH events
// belong to the actual break bar, never backdated.
// ============================================================================

// ── A. Inputs ────────────────────────────────────────────────────────────────
string GRP_STR = "Market Structure"
bool   showStructure       = input.bool(true,  "Show market structure", group = GRP_STR)
bool   showStructureLabels = input.bool(true,  "Show HH/HL/LH/LL labels", group = GRP_STR)
bool   showBOS             = input.bool(true,  "Show BOS", group = GRP_STR)
bool   showCHoCH           = input.bool(true,  "Show CHoCH", group = GRP_STR)
int    structureLeft       = input.int(5, "Structure pivot left bars", minval = 1, group = GRP_STR)
int    structureRight      = input.int(5, "Structure pivot right bars", minval = 1, group = GRP_STR)
string structureBreakMode  = input.string("Close", "Structure break confirmation", options = ["Close", "Wick"], group = GRP_STR)
bool   showStructureBias   = input.bool(false, "Show structure bias", group = GRP_STR)
int    maxStructureEvents  = input.int(50, "Max structure events", minval = 10, maxval = 200, group = GRP_STR)
bool   structureDebug      = input.bool(false, "Structure debug mode", group = GRP_STR)

// ── B. Structural pivot detection (global scope, every bar) ──────────────────
float strPivotHi = ta.pivothigh(high, structureLeft, structureRight)
float strPivotLo = ta.pivotlow(low, structureLeft, structureRight)

// ── C. Structural state (persistent across bars) ─────────────────────────────
var float prevStrHigh        = na
var float prevStrLow         = na
var float activeStrHigh      = na
var int   activeStrHighBar   = na
var bool  activeStrHighUsed  = false
var float activeStrLow       = na
var int   activeStrLowBar    = na
var bool  activeStrLowUsed   = false
var int   structureBias      = 0

// ── D. Object arrays for visualization (capped by maxStructureEvents) ────────
var array<line>  strEventLines  = array.new<line>()
var array<label> strEventLabels = array.new<label>()
var array<label> swingClassLabels = array.new<label>()

// Per-bar events (reset each bar; used by debug plotshapes).
bool strHiConfirmed = false
bool strLoConfirmed = false
bool bullBreakEvent = false
bool bearBreakEvent = false

if processBar
    // ── E. Break detection FIRST (uses current active levels before any new
    //       pivot can replace them, preventing same-bar lookahead) ─────────────
    float breakProbeHi = structureBreakMode == "Close" ? close : high
    float breakProbeLo = structureBreakMode == "Close" ? close : low
    bool  bullBreak = not na(activeStrHigh) and not activeStrHighUsed and breakProbeHi > activeStrHigh
    bool  bearBreak = not na(activeStrLow)  and not activeStrLowUsed  and breakProbeLo < activeStrLow

    // Same-bar double break (both sides breached): ambiguous — skip both.
    if bullBreak and bearBreak
        bullBreak := false
        bearBreak := false

    // ── F. Bullish break: BOS (with trend) or CHoCH (against trend) ──────────
    if bullBreak
        bullBreakEvent := true
        bool isCHoCH = structureBias == -1
        activeStrHighUsed := true
        if isCHoCH
            structureBias := 1
        else if structureBias == 0
            structureBias := 1
        if dmStructVis and showStructure and (isCHoCH ? showCHoCH : showBOS)
            string txt = isCHoCH ? "CHoCH" : "BOS"
            color  col = #089981
            line   ln  = line.new(activeStrHighBar, activeStrHigh, bar_index, activeStrHigh, xloc = xloc.bar_index, color = col, width = 1, style = isCHoCH ? line.style_dashed : line.style_solid)
            label  lbl = label.new(bar_index, activeStrHigh, txt, xloc = xloc.bar_index, style = label.style_none, textcolor = col, size = size.small)
            array.push(strEventLines, ln)
            array.push(strEventLabels, lbl)
            if array.size(strEventLines) > maxStructureEvents
                line.delete(array.shift(strEventLines))
            if array.size(strEventLabels) > maxStructureEvents
                label.delete(array.shift(strEventLabels))

    // ── G. Bearish break: BOS (with trend) or CHoCH (against trend) ──────────
    if bearBreak
        bearBreakEvent := true
        bool isCHoCH = structureBias == 1
        activeStrLowUsed := true
        if isCHoCH
            structureBias := -1
        else if structureBias == 0
            structureBias := -1
        if dmStructVis and showStructure and (isCHoCH ? showCHoCH : showBOS)
            string txt = isCHoCH ? "CHoCH" : "BOS"
            color  col = #f23645
            line   ln  = line.new(activeStrLowBar, activeStrLow, bar_index, activeStrLow, xloc = xloc.bar_index, color = col, width = 1, style = isCHoCH ? line.style_dashed : line.style_solid)
            label  lbl = label.new(bar_index, activeStrLow, txt, xloc = xloc.bar_index, style = label.style_none, textcolor = col, size = size.small)
            array.push(strEventLines, ln)
            array.push(strEventLabels, lbl)
            if array.size(strEventLines) > maxStructureEvents
                line.delete(array.shift(strEventLines))
            if array.size(strEventLabels) > maxStructureEvents
                label.delete(array.shift(strEventLabels))

    // ── H. New structural high: classify HH/LH + update active level ─────────
    if not na(strPivotHi)
        strHiConfirmed := true
        int hiOrigBar = bar_index - structureRight
        int hiClass   = na(prevStrHigh) ? 0 : (strPivotHi > prevStrHigh ? 1 : (strPivotHi < prevStrHigh ? -1 : 0))
        if dmStructVis and showStructure and showStructureLabels and hiClass != 0
            string hiText = hiClass == 1 ? "HH" : "LH"
            color  hiCol  = hiClass == 1 ? #089981 : #f23645
            label  lbl    = label.new(hiOrigBar, strPivotHi, hiText, xloc = xloc.bar_index, style = label.style_none, textcolor = hiCol, size = size.tiny)
            array.push(swingClassLabels, lbl)
            if array.size(swingClassLabels) > maxStructureEvents
                label.delete(array.shift(swingClassLabels))
        prevStrHigh       := strPivotHi
        activeStrHigh     := strPivotHi
        activeStrHighBar  := hiOrigBar
        activeStrHighUsed := false

    // ── I. New structural low: classify HL/LL + update active level ──────────
    if not na(strPivotLo)
        strLoConfirmed := true
        int loOrigBar = bar_index - structureRight
        int loClass   = na(prevStrLow) ? 0 : (strPivotLo > prevStrLow ? 1 : (strPivotLo < prevStrLow ? -1 : 0))
        if dmStructVis and showStructure and showStructureLabels and loClass != 0
            string loText = loClass == 1 ? "HL" : "LL"
            color  loCol  = loClass == 1 ? #089981 : #f23645
            label  lbl    = label.new(loOrigBar, strPivotLo, loText, xloc = xloc.bar_index, style = label.style_none, textcolor = loCol, size = size.tiny)
            array.push(swingClassLabels, lbl)
            if array.size(swingClassLabels) > maxStructureEvents
                label.delete(array.shift(swingClassLabels))
        prevStrLow       := strPivotLo
        activeStrLow     := strPivotLo
        activeStrLowBar  := loOrigBar
        activeStrLowUsed := false

// ── J. Structure bias display (small table, top-right corner) ────────────────
var table biasTable = (showStructureBias and dmBiasTbl) ? table.new(position.top_right, 1, 1, bgcolor = color.new(#000000, 100)) : na
if showStructureBias and dmBiasTbl and barstate.islast and not na(biasTable)
    string biasText = structureBias == 1 ? "Bullish" : structureBias == -1 ? "Bearish" : "Neutral"
    color  biasCol  = structureBias == 1 ? #089981 : structureBias == -1 ? #f23645 : #787b86
    table.cell(biasTable, 0, 0, biasText, text_color = biasCol, text_size = size.small)

// ============================================================================
// PHASE 4 — LIQUIDITY ENGINE
// Identifies BSL/SSL, equal highs/lows, liquidity sweeps (wick grab + close
// rejection), and consumed levels (close through). Detection only — no signals.
// Uses its own confirmed pivots independent from Phase 2 and Phase 3.
// Non-repainting: levels only become active after pivot confirmation.
// Processing order: check existing levels FIRST, then add new pivots.
// ============================================================================

// ── A. Inputs ────────────────────────────────────────────────────────────────
string GRP_LE = "Liquidity Engine"
bool   showLiqEngine  = input.bool(true,  "Show liquidity", group = GRP_LE)
bool   showLeBsl      = input.bool(true,  "Show BSL/SSL", group = GRP_LE)
bool   showLeEqual    = input.bool(true,  "Show equal highs/lows", group = GRP_LE)
bool   showLeSweeps   = input.bool(true,  "Show liquidity sweeps", group = GRP_LE)
int    lePivotLeft    = input.int(5,  "Liquidity pivot left bars", minval = 1, group = GRP_LE)
int    lePivotRight   = input.int(5,  "Liquidity pivot right bars", minval = 1, group = GRP_LE)
int    leEqTolTicks   = input.int(4,  "Equal level tolerance (ticks)", minval = 1, group = GRP_LE, tooltip = "Two same-side pivots within this many ticks are an equal high/low cluster.")
int    leMaxLevels    = input.int(100, "Max liquidity levels", minval = 10, maxval = 500, group = GRP_LE, tooltip = "Per side (BSL / SSL). Oldest levels removed when exceeded.")
bool   leDebug        = input.bool(false, "Liquidity debug mode", group = GRP_LE)

// ── B. Pivot detection (global scope, every bar) ─────────────────────────────
float lePivotHi = ta.pivothigh(high, lePivotLeft, lePivotRight)
float lePivotLo = ta.pivotlow(low, lePivotLeft, lePivotRight)

// ── C. Liquidity level model ─────────────────────────────────────────────────
type LeLevel
    float price
    int   origBar
    int   confirmBar
    bool  isBuy
    bool  isEqual
    bool  swept
    bool  consumed
    line  ln

var array<LeLevel> leBslLevels    = array.new<LeLevel>()
var array<LeLevel> leSslLevels    = array.new<LeLevel>()
var array<line>    leRetiredLines = array.new<line>()

var float lePrevSwingHigh = na
var float lePrevSwingLow  = na

float leEqTol = leEqTolTicks * syminfo.mintick

// Per-bar events (reset each bar; used for markers and debug).
bool leBslSweepEvt    = false
bool leSslSweepEvt    = false
bool leBslConsumedEvt = false
bool leSslConsumedEvt = false
bool leBslCreatedEvt  = false
bool leSslCreatedEvt  = false
bool leEqhEvt         = false
bool leEqlEvt         = false

if processBar
    // ── D. Check active BSL levels for sweep / consumption ───────────────────
    //    Sweep  = wick grabbed above + close rejected below (high > P, close < P)
    //    Consumed = close through (close > P)
    if array.size(leBslLevels) > 0
        for i = array.size(leBslLevels) - 1 to 0
            LeLevel lvl = array.get(leBslLevels, i)
            if not na(lvl.ln)
                line.set_x2(lvl.ln, bar_index)
            bool eligible = bar_index > lvl.confirmBar
            if eligible
                if high > lvl.price and close < lvl.price
                    lvl.swept := true
                    leBslSweepEvt := true
                    if not na(lvl.ln)
                        line.set_extend(lvl.ln, extend.none)
                        line.set_x2(lvl.ln, bar_index)
                        line.set_color(lvl.ln, sweptLiqColor)
                        line.set_style(lvl.ln, line.style_dashed)
                        array.push(leRetiredLines, lvl.ln)
                        if array.size(leRetiredLines) > leMaxLevels
                            line.delete(array.shift(leRetiredLines))
                    array.remove(leBslLevels, i)
                else if close > lvl.price
                    lvl.consumed := true
                    leBslConsumedEvt := true
                    if not na(lvl.ln)
                        line.set_extend(lvl.ln, extend.none)
                        line.set_x2(lvl.ln, bar_index)
                        line.set_color(lvl.ln, sweptLiqColor)
                        line.set_style(lvl.ln, line.style_dotted)
                        array.push(leRetiredLines, lvl.ln)
                        if array.size(leRetiredLines) > leMaxLevels
                            line.delete(array.shift(leRetiredLines))
                    array.remove(leBslLevels, i)

    // ── E. Check active SSL levels for sweep / consumption ───────────────────
    //    Sweep  = wick grabbed below + close rejected above (low < P, close > P)
    //    Consumed = close through (close < P)
    if array.size(leSslLevels) > 0
        for i = array.size(leSslLevels) - 1 to 0
            LeLevel lvl = array.get(leSslLevels, i)
            if not na(lvl.ln)
                line.set_x2(lvl.ln, bar_index)
            bool eligible = bar_index > lvl.confirmBar
            if eligible
                if low < lvl.price and close > lvl.price
                    lvl.swept := true
                    leSslSweepEvt := true
                    if not na(lvl.ln)
                        line.set_extend(lvl.ln, extend.none)
                        line.set_x2(lvl.ln, bar_index)
                        line.set_color(lvl.ln, sweptLiqColor)
                        line.set_style(lvl.ln, line.style_dashed)
                        array.push(leRetiredLines, lvl.ln)
                        if array.size(leRetiredLines) > leMaxLevels
                            line.delete(array.shift(leRetiredLines))
                    array.remove(leSslLevels, i)
                else if close < lvl.price
                    lvl.consumed := true
                    leSslConsumedEvt := true
                    if not na(lvl.ln)
                        line.set_extend(lvl.ln, extend.none)
                        line.set_x2(lvl.ln, bar_index)
                        line.set_color(lvl.ln, sweptLiqColor)
                        line.set_style(lvl.ln, line.style_dotted)
                        array.push(leRetiredLines, lvl.ln)
                        if array.size(leRetiredLines) > leMaxLevels
                            line.delete(array.shift(leRetiredLines))
                    array.remove(leSslLevels, i)

    // ── F. Process newly confirmed pivot highs → BSL levels ──────────────────
    if not na(lePivotHi)
        int hiOrigBar = bar_index - lePivotRight
        bool isDupHi = false
        if array.size(leBslLevels) > 0
            for j = 0 to array.size(leBslLevels) - 1
                if math.abs(array.get(leBslLevels, j).price - lePivotHi) < syminfo.mintick
                    isDupHi := true
                    break
        if not isDupHi
            leBslCreatedEvt := true
            bool isEqH = not na(lePrevSwingHigh) and math.abs(lePivotHi - lePrevSwingHigh) <= leEqTol
            if isEqH
                leEqhEvt := true
            line hiLn = na
            if dmLiqVis and showLiqEngine and showLeBsl
                color hiCol = (isEqH and showLeEqual) ? eqColor : bslColor
                hiLn := line.new(bar_index, lePivotHi, bar_index + 1, lePivotHi, xloc = xloc.bar_index, extend = extend.right, color = hiCol, width = 1)
            array.push(leBslLevels, LeLevel.new(lePivotHi, hiOrigBar, bar_index, true, isEqH, false, false, hiLn))
            if array.size(leBslLevels) > leMaxLevels
                LeLevel old = array.shift(leBslLevels)
                if not na(old.ln)
                    line.delete(old.ln)
        lePrevSwingHigh := lePivotHi

    // ── G. Process newly confirmed pivot lows → SSL levels ───────────────────
    if not na(lePivotLo)
        int loOrigBar = bar_index - lePivotRight
        bool isDupLo = false
        if array.size(leSslLevels) > 0
            for j = 0 to array.size(leSslLevels) - 1
                if math.abs(array.get(leSslLevels, j).price - lePivotLo) < syminfo.mintick
                    isDupLo := true
                    break
        if not isDupLo
            leSslCreatedEvt := true
            bool isEqL = not na(lePrevSwingLow) and math.abs(lePivotLo - lePrevSwingLow) <= leEqTol
            if isEqL
                leEqlEvt := true
            line loLn = na
            if dmLiqVis and showLiqEngine and showLeBsl
                color loCol = (isEqL and showLeEqual) ? eqColor : sslColor
                loLn := line.new(bar_index, lePivotLo, bar_index + 1, lePivotLo, xloc = xloc.bar_index, extend = extend.right, color = loCol, width = 1)
            array.push(leSslLevels, LeLevel.new(lePivotLo, loOrigBar, bar_index, false, isEqL, false, false, loLn))
            if array.size(leSslLevels) > leMaxLevels
                LeLevel old = array.shift(leSslLevels)
                if not na(old.ln)
                    line.delete(old.ln)
        lePrevSwingLow := lePivotLo

// ── H. Sweep markers (normal mode) ──────────────────────────────────────────
plotshape(dmLiqVis and showLiqEngine and showLeSweeps and leBslSweepEvt, title = "LE: BSL Sweep", style = shape.triangledown, location = location.abovebar, color = sweepColor, size = size.tiny)
plotshape(dmLiqVis and showLiqEngine and showLeSweeps and leSslSweepEvt, title = "LE: SSL Sweep", style = shape.triangleup,   location = location.belowbar, color = sweepColor, size = size.tiny)

// ============================================================================
// PHASE 5 — CRT CORE CONFLUENCE / SETUP QUALITY ENGINE
// Scores potential LONG/SHORT setups by confluence of liquidity sweeps, market
// structure (BOS/CHoCH), bias, displacement, and session timing.
// Detection and scoring only — no automatic entries or strategy orders.
// Scoring: Liquidity 25 + Structure 30 + Bias 20 + Displacement 15 + Session 10 = 100
// ============================================================================

// ── A. Inputs ────────────────────────────────────────────────────────────────
string GRP_SE = "Setup Engine"
bool   seEnabled         = input.bool(true,  "Enable setup engine", group = GRP_SE)
int    seMinScore        = input.int(70,  "Minimum setup score", minval = 0, maxval = 100, group = GRP_SE)
int    seStrongScore     = input.int(85,  "Strong setup score", minval = 0, maxval = 100, group = GRP_SE)
int    seLiqLookback     = input.int(20,  "Liquidity lookback bars", minval = 5, maxval = 100, group = GRP_SE, tooltip = "How far back to look for a qualifying liquidity sweep.")
int    seDispLookback    = input.int(10,  "Displacement lookback", minval = 3, maxval = 50, group = GRP_SE, tooltip = "Lookback for average candle body used in displacement test.")
float  seDispMultiplier  = input.float(1.5, "Displacement body multiplier", minval = 1.0, step = 0.1, group = GRP_SE, tooltip = "Current body must exceed this multiple of the average body to count as displacement.")
bool   seStrictSession   = input.bool(false, "Strict session filter", group = GRP_SE, tooltip = "On: setups outside the session window are rejected entirely. Off: reduced score only.")
string seSessionTime     = input.session("0930-1600", "Session window", group = GRP_SE, tooltip = "Time window for full session quality points (exchange timezone).")
bool   seAntiChase       = input.bool(true,  "Enable anti-chase filter", group = GRP_SE, tooltip = "Reject setups when price has already moved excessively from the structural level.")
float  seChaseAtrMax     = input.float(3.0, "Max chase distance (x ATR)", minval = 1.0, step = 0.5, group = GRP_SE)
int    seCooldown        = input.int(5, "Setup cooldown bars", minval = 0, maxval = 50, group = GRP_SE, tooltip = "Minimum bars between consecutive same-side setups.")
bool   showSetupMarkers  = input.bool(true,  "Show setup markers", group = GRP_SE)
bool   showSetupScore    = input.bool(true,  "Show setup score", group = GRP_SE)
bool   showSetupDebug    = input.bool(false, "Show setup debug table", group = GRP_SE)

// ── B. Pre-computed series (global scope, every bar) ─────────────────────────
float seAvgBody   = ta.sma(math.abs(close - open), seDispLookback)
float seAtr14     = ta.atr(14)
bool  seInSession = not na(time(timeframe.period, seSessionTime))
bool  seBullDisp  = close > open and not na(seAvgBody) and seAvgBody > 0 and (close - open) > seDispMultiplier * seAvgBody
bool  seBearDisp  = close < open and not na(seAvgBody) and seAvgBody > 0 and (open - close) > seDispMultiplier * seAvgBody

// ── C. Tracking state (persistent) ───────────────────────────────────────────
var int   seBullStructBar   = na
var bool  seBullIsCHoCH     = false
var int   seBearStructBar   = na
var bool  seBearIsCHoCH     = false
var int   seSslSweepBar     = na
var int   seBslSweepBar     = na
var int   seLongCoolUntil   = 0
var int   seShortCoolUntil  = 0
var string seLongType       = ""
var string seShortType      = ""
var int   seLongScoreVal    = 0
var int   seShortScoreVal   = 0

var array<label> seLabels = array.new<label>()

// Per-bar setup events (reset each bar).
bool seLongEvt     = false
bool seShortEvt    = false
bool seLongStrong  = false
bool seShortStrong = false

if processBar and seEnabled
    // Update tracking from Phase 3 structure events.
    // structureBias[1] = bias BEFORE this bar's break (Phase 3 already updated it).
    if bullBreakEvent
        seBullStructBar := bar_index
        seBullIsCHoCH   := structureBias[1] == -1
    if bearBreakEvent
        seBearStructBar := bar_index
        seBearIsCHoCH   := structureBias[1] == 1

    // Update tracking from Phase 4 liquidity sweep events.
    if leSslSweepEvt
        seSslSweepBar := bar_index
    if leBslSweepEvt
        seBslSweepBar := bar_index

    // ── D. Score components ──────────────────────────────────────────────────
    // LIQUIDITY (25 pts): recent sweep on the correct side
    bool  recentSsl = not na(seSslSweepBar) and bar_index - seSslSweepBar <= seLiqLookback
    bool  recentBsl = not na(seBslSweepBar) and bar_index - seBslSweepBar <= seLiqLookback
    float longLiq   = recentSsl ? 25.0 : 0.0
    float shortLiq  = recentBsl ? 25.0 : 0.0

    // STRUCTURE (30 pts): recent BOS/CHoCH on the correct side
    bool  recentBullStr = not na(seBullStructBar) and bar_index - seBullStructBar <= seLiqLookback
    bool  recentBearStr = not na(seBearStructBar) and bar_index - seBearStructBar <= seLiqLookback
    float longStr  = recentBullStr ? (seBullIsCHoCH ? 30.0 : (structureBias == 1 ? 30.0 : 20.0)) : 0.0
    float shortStr = recentBearStr ? (seBearIsCHoCH ? 30.0 : (structureBias == -1 ? 30.0 : 20.0)) : 0.0

    // BIAS (20 pts): current structure bias alignment
    float longBias  = structureBias == 1 ? 20.0 : structureBias == 0 ? 10.0 : 5.0
    float shortBias = structureBias == -1 ? 20.0 : structureBias == 0 ? 10.0 : 5.0

    // DISPLACEMENT (15 pts): meaningful candle body on this bar
    float longDisp  = seBullDisp ? 15.0 : 0.0
    float shortDisp = seBearDisp ? 15.0 : 0.0

    // SESSION (10 pts): within preferred trading window
    float longSess  = seInSession ? 10.0 : 0.0
    float shortSess = seInSession ? 10.0 : 0.0

    // Total (clamped 0–100)
    float lScore = math.min(longLiq + longStr + longBias + longDisp + longSess, 100.0)
    float sScore = math.min(shortLiq + shortStr + shortBias + shortDisp + shortSess, 100.0)

    // ANTI-CHASE: reject if price already moved too far from the structural level
    if seAntiChase and not na(seAtr14) and seAtr14 > 0
        if not na(activeStrHigh) and close > activeStrHigh and (close - activeStrHigh) > seChaseAtrMax * seAtr14
            lScore := 0.0
        if not na(activeStrLow) and close < activeStrLow and (activeStrLow - close) > seChaseAtrMax * seAtr14
            sScore := 0.0

    // STRICT SESSION: hard reject outside window
    if seStrictSession and not seInSession
        lScore := 0.0
        sScore := 0.0

    seLongScoreVal  := int(lScore)
    seShortScoreVal := int(sScore)

    // ── E. Trigger with duplicate prevention ─────────────────────────────────
    // Require a NEW event (structure break or liquidity sweep) on THIS bar
    // plus cooldown expired, plus score above minimum.
    bool newLongEvt  = bullBreakEvent or leSslSweepEvt
    bool newShortEvt = bearBreakEvent or leBslSweepEvt

    if bar_index >= seLongCoolUntil and newLongEvt and lScore >= seMinScore
        seLongEvt       := true
        seLongStrong    := lScore >= seStrongScore
        seLongCoolUntil := bar_index + seCooldown
        seLongType      := seBullIsCHoCH and recentBullStr and recentSsl ? "Reversal" : "Continuation"
        if showSetupScore
            string txt = "L " + str.tostring(int(lScore))
            color  col = seLongStrong ? #00e676 : #089981
            label  lbl = label.new(bar_index, low, txt, xloc = xloc.bar_index, style = label.style_none, textcolor = col, size = size.tiny)
            array.push(seLabels, lbl)
            if array.size(seLabels) > 50
                label.delete(array.shift(seLabels))

    if bar_index >= seShortCoolUntil and newShortEvt and sScore >= seMinScore
        seShortEvt       := true
        seShortStrong    := sScore >= seStrongScore
        seShortCoolUntil := bar_index + seCooldown
        seShortType      := seBearIsCHoCH and recentBearStr and recentBsl ? "Reversal" : "Continuation"
        if showSetupScore
            string txt = "S " + str.tostring(int(sScore))
            color  col = seShortStrong ? #ff1744 : #f23645
            label  lbl = label.new(bar_index, high, txt, xloc = xloc.bar_index, style = label.style_none, textcolor = col, size = size.tiny)
            array.push(seLabels, lbl)
            if array.size(seLabels) > 50
                label.delete(array.shift(seLabels))

// ── F. Setup markers ─────────────────────────────────────────────────────────
plotshape(seEnabled and showSetupMarkers and seLongEvt,  title = "Long Setup",  style = shape.triangleup,   location = location.belowbar, color = seLongStrong ? #00e676 : #089981, size = size.tiny)
plotshape(seEnabled and showSetupMarkers and seShortEvt, title = "Short Setup", style = shape.triangledown, location = location.abovebar, color = seShortStrong ? #ff1744 : #f23645, size = size.tiny)

// ── G. Debug table (bottom-right, current state only) ────────────────────────
var table seDebugTbl = (showSetupDebug and dmDiag) ? table.new(position.bottom_right, 2, 10, bgcolor = color.new(#1e1e1e, 20), border_color = color.new(#444444, 50), border_width = 1) : na
if showSetupDebug and dmDiag and barstate.islast and not na(seDebugTbl)
    color _h = #aaaaaa
    table.cell(seDebugTbl, 0, 0, "Bias",             text_color = _h, text_size = size.tiny)
    table.cell(seDebugTbl, 1, 0, structureBias == 1 ? "Bullish" : structureBias == -1 ? "Bearish" : "Neutral", text_color = structureBias == 1 ? #089981 : structureBias == -1 ? #f23645 : #787b86, text_size = size.tiny)
    table.cell(seDebugTbl, 0, 1, "SSL sweep",        text_color = _h, text_size = size.tiny)
    table.cell(seDebugTbl, 1, 1, not na(seSslSweepBar) and bar_index - seSslSweepBar <= seLiqLookback ? str.tostring(bar_index - seSslSweepBar) + " bars" : "—", text_color = _h, text_size = size.tiny)
    table.cell(seDebugTbl, 0, 2, "BSL sweep",        text_color = _h, text_size = size.tiny)

    array.set(vc,   b, array.get(vc,   b) + 1)
    array.set(vr3,  b, array.get(vr3,  b) + nz(r3))

    array.set(vr12, b, array.get(vr12, b) + nz(r12))
    array.set(vmfe, b, array.get(vmfe, b) + mf)
    array.set(vmae, b, array.get(vmae, b) + ma)
    if nz(r3)  > 0
        array.set(vp3,  b, array.get(vp3,  b) + 1)
    if nz(r6)  > 0
        array.set(vp6,  b, array.get(vp6,  b) + 1)
    if nz(r12) > 0
        array.set(vp12, b, array.get(vp12, b) + 1)

                    if array.get(valDays, d) == dayId

                        break
            if not dayExL

    if seShortEvt and valInRange

            valAccepted += 1

            bool dayExS = false

    if array.size(valPending) > 0

            ValSignal sig = array.get(valPending, i)
            int elapsed = bar_index - sig.sigBar

            // MFE / MAE (bars 1–12 after signal)
            if elapsed >= 1 and elapsed <= 12
                if sig.dir == 1
                    sig.mfe := math.max(sig.mfe, high - sig.sigPrice)
                    sig.mae := math.max(sig.mae, sig.sigPrice - low)
                else
                    sig.mfe := math.max(sig.mfe, sig.sigPrice - low)
                    sig.mae := math.max(sig.mae, high - sig.sigPrice)

            // Milestone returns
            if elapsed == 3 and na(sig.ret3)
                sig.ret3 := sig.dir == 1 ? close - sig.sigPrice : sig.sigPrice - close
            if elapsed == 6 and na(sig.ret6)
                sig.ret6 := sig.dir == 1 ? close - sig.sigPrice : sig.sigPrice - close

                int b0 = sig.dir == 1 ? 0 : 1
                int b1 = sig.score >= 85 ? (sig.dir == 1 ? 4 : 7) : sig.score >= 75 ? (sig.dir == 1 ? 3 : 6) : (sig.dir == 1 ? 2 : 5)

                addBucket(b1, sig.ret3, sig.ret6, sig.ret12, sig.mfe, sig.mae)
                addBucket(b2, sig.ret3, sig.ret6, sig.ret12, sig.mfe, sig.mae)

                    string vTxt = dCh + str.tostring(sig.score) + "\n60m " + (nz(sig.ret12) >= 0 ? "+" : "") + str.tostring(math.round(nz(sig.ret12) * 10) / 10) + "\nMFE " + str.tostring(math.round(sig.mfe * 10) / 10) + "\nMAE " + str.tostring(math.round(sig.mae * 10) / 10)
                    color  vCol = sig.dir == 1 ? #089981 : #f23645
                    float  vY   = sig.dir == 1 ? nz(low[off], close) : nz(high[off], close)
                    label  vLbl = label.new(sig.sigBar, vY, vTxt, xloc = xloc.bar_index, style = label.style_none, textcolor = vCol, size = size.tiny)

                    gp += rr

p14ProcessFunnel() =>
    int state = array.get(p14fst, 0)
    int dir = array.get(p14fdr, 0)

    int bosBar = array.get(p14fbb, 0)
    int retestBar = array.get(p14frb, 0)
    int score = array.get(p14fsc, 0)
    float bosLvl = array.get(p14fbl, 0)
    int hm = hour * 60 + minute
    int tmb = hm >= 1080 or hm < 240 ? 0 : hm < 570 ? 1 : hm < 630 ? 2 : hm < 720 ? 3 : hm < 840 ? 4 : hm < 960 ? 5 : hm < 1080 ? 6 : 0
    bool cFilter = p7HtfRegime != 0 and tmb != 6
    bool hasSetup = (seLongEvt or seShortEvt) and cFilter

    if hasSetup and state == 0
        state := 1
        dir := seLongEvt ? 1 : -1
        setupBar := bar_index
        bosBar := na
        retestBar := na
        bosLvl := na
        score := dir == 1 ? seLongScoreVal : seShortScoreVal
        array.set(p14fc, 0, array.get(p14fc, 0) + 1)
        openP14(0, dir, score)

    if state == 1
        bool bosOk = (dir == 1 and bullBreakEvent) or (dir == -1 and bearBreakEvent)
        bool oppBos = (dir == 1 and bearBreakEvent) or (dir == -1 and bullBreakEvent)
        if bosOk
            float brokenLvl = dir == 1 ? activeStrHigh[1] : activeStrLow[1]
            bosLvl := not na(brokenLvl) ? brokenLvl : (dir == 1 ? activeStrHigh : activeStrLow)
            bosBar := bar_index
            array.set(p14fc, 1, array.get(p14fc, 1) + 1)
            openP14(1, dir, score)
            state := 2
        else if oppBos or bar_index - setupBar > p12Expiry
            state := 0

    else if state == 2 and not na(bosLvl)
        float tol = nz(seAtr14, 1.0) * p12RetestTol
        bool eligibleRetest = not na(bosBar) and bar_index > bosBar
        bool retestTouch = eligibleRetest and (dir == 1 ? low <= bosLvl + tol : high >= bosLvl - tol)
        bool retInv = eligibleRetest and (dir == 1 ? close < bosLvl - tol : close > bosLvl + tol)
        if retInv
            state := 0
        else if retestTouch
            retestBar := bar_index
            array.set(p14fc, 2, array.get(p14fc, 2) + 1)
            openP14(2, dir, score)
            state := 3
        else if not na(bosBar) and bar_index - bosBar > p12Expiry
            state := 0

    else if state == 3 and not na(bosLvl) and not na(retestBar)
        float tol = nz(seAtr14, 1.0) * p12RetestTol
        bool eligibleConfirm = bar_index > retestBar
        bool confirmed = eligibleConfirm and (dir == 1 ? (close > open and close > bosLvl) : (close < open and close < bosLvl))
        bool confInv = eligibleConfirm and (dir == 1 ? close < bosLvl - tol : close > bosLvl + tol)
        if confirmed
            array.set(p14fc, 3, array.get(p14fc, 3) + 1)
            openP14(3, dir, score)
            state := 4
        else if confInv or bar_index - retestBar > p12Expiry
            state := 0

    if state == 4
        state := 0

    array.set(p14fst, 0, state)
    array.set(p14fdr, 0, dir)
    array.set(p14fsb, 0, setupBar)
    array.set(p14fbb, 0, bosBar)
    array.set(p14frb, 0, retestBar)
    array.set(p14fsc, 0, score)
    array.set(p14fbl, 0, bosLvl)

        array.set(p7c, b, array.get(p7c, b) + 1)
        if rR > 0
            array.set(p7w, b, array.get(p7w, b) + 1)
            array.set(p7gp, b, array.get(p7gp, b) + rR)
        if rR < 0
            array.set(p7gl, b, array.get(p7gl, b) + math.abs(rR))
        array.set(p7rs, b, array.get(p7rs, b) + rR)

    float sr = array.get(p7rs, b)
    int   w  = array.get(p7w, b)

    float gp = array.get(p7gp, b)
    float gl = array.get(p7gl, b)
    float pf = gl > 0 ? gp / gl : (gp > 0 ? 99.9 : 0.0)

    table.cell(t, 3, row, str.tostring(math.round(ar * 100) / 100), text_color = ar >= 0 ? #089981 : #f23645,  text_size = size.tiny)
    table.cell(t, 4, row, str.tostring(math.round(sr * 100) / 100), text_color = sr >= 0 ? #089981 : #f23645,  text_size = size.tiny)
    table.cell(t, 5, row, str.tostring(math.round(pf * 10) / 10),   text_color = pf >= 1 ? #089981 : #f23645,  text_size = size.tiny)

if processBar and trEnabled

    if trActive

// ── Phase 12 BOS + retest + confirmation state machine (research only) ───────
// Full research funnel always runs:
// SETUP → BOS → RETEST → CONFIRM.
// Phase 13 hooks open each comparison model at the appropriate confirmed stage.
// This does NOT change Phase 11 live logic.
if processBar and p12Enabled
    bool p12hasSetup = (seLongEvt or seShortEvt) and liveFilter

    // Start a new parent candidate only while the research funnel is idle.
    if p12hasSetup and p12State == 0
        p12State     := 1   // WAIT_BOS
        p12Dir       := seLongEvt ? 1 : -1
        p12SetupBar  := bar_index
        p12BosLvl    := na
        p12BosBar    := na
        p12RetestBar := na
        p12Setups    += 1
        if p13Enabled
            openP13(0, p12Dir)   // CONTROL entry

    if p12State == 1
        // WAIT_BOS: confirmed structure break in setup direction.
        bool bosOk  = (p12Dir == 1 and bullBreakEvent) or (p12Dir == -1 and bearBreakEvent)
        bool oppBos = (p12Dir == 1 and bearBreakEvent) or (p12Dir == -1 and bullBreakEvent)

        if bosOk
            // Preserve the structure level that was actually broken.
            // Phase 3 can update activeStrHigh/Low later on this same bar,
            // so [1] avoids accidentally storing a newly-confirmed pivot.
            float brokenLvl = p12Dir == 1 ? activeStrHigh[1] : activeStrLow[1]
            p12BosLvl  := not na(brokenLvl) ? brokenLvl : (p12Dir == 1 ? activeStrHigh : activeStrLow)
            p12BosBar  := bar_index
            p12BosConf += 1
            if p13Enabled
                openP13(1, p12Dir)   // BOS entry
            p12State := 2            // ALWAYS continue to WAIT_RETEST

        else if oppBos
            p12InvBos += 1
            p12State  := 0

        else if bar_index - p12SetupBar > p12Expiry
            p12ExpBos += 1
            p12State  := 0

    else if p12State == 2 and not na(p12BosLvl)
        // WAIT_RETEST: retest can only happen AFTER the BOS bar.
        float tol = nz(seAtr14, 1.0) * p12RetestTol
        bool eligibleRetest = not na(p12BosBar) and bar_index > p12BosBar
        bool retestTouch = eligibleRetest and (p12Dir == 1 ? low <= p12BosLvl + tol : high >= p12BosLvl - tol)
        bool retInv      = eligibleRetest and (p12Dir == 1 ? close < p12BosLvl - tol : close > p12BosLvl + tol)

        if retInv
            p12InvRet += 1
            p12State  := 0

        else if retestTouch
            p12RetestOk  += 1
            p12RetestBar := bar_index
            if p13Enabled
                openP13(2, p12Dir)   // RETEST entry
            p12State := 3            // ALWAYS continue to WAIT_CONFIRM

        else if not na(p12BosBar) and bar_index - p12BosBar > p12Expiry
            p12ExpRet += 1
            p12State  := 0

    else if p12State == 3 and not na(p12BosLvl) and not na(p12RetestBar)
        // WAIT_CONFIRM: confirmation must occur on a bar AFTER the retest.
        // Long: bullish candle closing above BOS.
        // Short: bearish candle closing below BOS.
        float cTol = nz(seAtr14, 1.0) * p12RetestTol
        bool eligibleConfirm = bar_index > p12RetestBar
        bool confirmed = eligibleConfirm and (p12Dir == 1 ? (close > open and close > p12BosLvl) : (close < open and close < p12BosLvl))
        bool confInv   = eligibleConfirm and (p12Dir == 1 ? close < p12BosLvl - cTol : close > p12BosLvl + cTol)

        if confirmed
            p12ConfirmOk += 1
            if p13Enabled
                openP13(3, p12Dir)   // CONFIRM entry
            p12State := 4

        else if confInv
            p12InvConf += 1
            p12State   := 0

        else if bar_index - p12RetestBar > p12Expiry
            p12ExpConf += 1
            p12State   := 0

    // READY = completed research candidate. Reset after counting/opening model.
    if p12State == 4
        p12State := 0

// ── Phase 13 bar-by-bar trade simulation for all 4 models ────────────────────
// Wrapped in a function to keep Pine's main script body below compiler limits.
p13ProcessTrades() =>
    for m = 0 to 3
        if array.get(p13act, m)
            int elapsed = bar_index - array.get(p13br, m)
            if elapsed >= 1
                int   d = array.get(p13dr, m)
                float s = array.get(p13st, m)
                float t = array.get(p13tg, m)
                float r = array.get(p13rk, m)
                float e = array.get(p13en, m)
                float rr = na
                if d == 1
                    if low <= s
                        rr := -1.0
                    else if high >= t
                        rr := trTargetR
                    else if elapsed >= trMaxBarsCnt
                        rr := r > 0 ? (close - e) / r : 0.0
                else
                    if high >= s
                        rr := -1.0
                    else if low <= t
                        rr := trTargetR
                    else if elapsed >= trMaxBarsCnt
                        rr := r > 0 ? (e - close) / r : 0.0
                if not na(rr)
                    closeP13(m, rr)

if processBar and p13Enabled
    p13ProcessTrades()

// ── Phase 14 date-scoped bar-by-bar simulation ──────────────────────────────
// Uses the same ATR entry/stop/target and duration rules as Phase 13. Stop is
// checked before target on ambiguous bars. Any position still open at the
// inclusive window end is marked to the final in-window close.
p14ProcessTrades() =>
    for m = 0 to 3
        if array.get(p14act, m)
            int elapsed = bar_index - array.get(p14bar, m)
            int d = array.get(p14dir, m)
            float r = array.get(p14rk, m)
            float e = array.get(p14en, m)
            float rr = na

            // If the chart jumps across the end boundary (for example a
            // session gap), use the preceding close and never the out-of-range
            // bar's high/low.
            if time >= p14EndExclusive
                float endPx = nz(close[1], close)
                rr := r > 0 ? (d == 1 ? (endPx - e) / r : (e - endPx) / r) : 0.0
            else if elapsed >= 1
                float s = array.get(p14st, m)
                float t = array.get(p14tg, m)
                if d == 1
                    if low <= s
                        rr := -1.0
                    else if high >= t
                        rr := trTargetR
                    else if elapsed >= trMaxBarsCnt
                        rr := r > 0 ? (close - e) / r : 0.0
                    else if time_close >= p14EndExclusive
                        rr := r > 0 ? (close - e) / r : 0.0
                else
                    if high >= s
                        rr := -1.0
                    else if low <= t
                        rr := trTargetR
                    else if elapsed >= trMaxBarsCnt
                        rr := r > 0 ? (e - close) / r : 0.0
                    else if time_close >= p14EndExclusive
                        rr := r > 0 ? (e - close) / r : 0.0

            if not na(rr)
                closeP14(m, rr)

if processBar and p14Enabled
    if p14InWindow
        p14ProcessFunnel()
    p14ProcessTrades()

        float wr = n > 0 ? w * 100.0 / n : 0.0
        float gp = array.get(p13gp, m)
        float gl = array.get(p13gl, m)
        float pf = gl > 0 ? gp / gl : (gp > 0 ? 99.9 : 0.0)
        float dd = array.get(p13dd, m)
        int ret = p12Setups > 0 ? math.round(n * 100.0 / p12Setups) : 0
        table.cell(tbl, 0, m + 1, array.get(names, m), text_color = d, text_size = size.tiny, text_halign = text.align_left)
        table.cell(tbl, 1, m + 1, str.tostring(n), text_color = d, text_size = size.tiny)
        table.cell(tbl, 2, m + 1, str.tostring(ret) + "%", text_color = d, text_size = size.tiny)
        table.cell(tbl, 3, m + 1, str.tostring(math.round(wr)) + "%", text_color = wr >= 50 ? #089981 : #f23645, text_size = size.tiny)
        table.cell(tbl, 4, m + 1, str.tostring(math.round(ar * 100) / 100), text_color = ar >= 0 ? #089981 : #f23645, text_size = size.tiny)
        table.cell(tbl, 5, m + 1, str.tostring(math.round(sr * 100) / 100), text_color = sr >= 0 ? #089981 : #f23645, text_size = size.tiny)
        table.cell(tbl, 6, m + 1, str.tostring(math.round(pf * 10) / 10), text_color = pf >= 1 ? #089981 : #f23645, text_size = size.tiny)
        table.cell(tbl, 7, m + 1, str.tostring(math.round(dd * 100) / 100), text_color = #f23645, text_size = size.tiny)

    int bestM = 0
    float bestAvg = array.get(p13n, 0) > 0 ? array.get(p13rs, 0) / array.get(p13n, 0) : -999.0
    for m = 1 to 3
        int mn = array.get(p13n, m)
        float ma = mn > 0 ? array.get(p13rs, m) / mn : -999.0
        if mn >= 20 and ma > bestAvg
            bestAvg := ma
            bestM := m
    string verdict = array.get(p13n, bestM) < 20 ? "INSUFFICIENT DATA" : array.get(names, bestM) + " (AvgR " + str.tostring(math.round(bestAvg * 100) / 100) + ")"
    table.cell(tbl, 0, 5, "Best", text_color = h, text_size = size.tiny, text_halign = text.align_left)
    table.cell(tbl, 1, 5, verdict, text_color = #ff9800, text_size = size.tiny)
    table.cell(tbl, 0, 6, "Repaint", text_color = d, text_size = size.tiny, text_halign = text.align_left)
    table.cell(tbl, 1, 6, "PASS", text_color = #089981, text_size = size.tiny)

p14Fmt(float value) =>
    str.tostring(math.round(value * 100.0) / 100.0)

p14BreakRow(table tbl, int row, string name, array<float> stats, int bucketCount, int bucket) =>
    color d = #aaaaaa
    table.cell(tbl, 0, row, name, text_color = d, text_size = size.tiny, text_halign = text.align_left)
    for m = 0 to 3
        int base = (m * bucketCount + bucket) * 5
        int n = int(array.get(stats, base))
        int wins = int(array.get(stats, base + 1))
        float totalR = array.get(stats, base + 2)
        float gp = array.get(stats, base + 3)
        float gl = array.get(stats, base + 4)
        float avgR = n > 0 ? totalR / n : 0.0
        float winPct = n > 0 ? wins * 100.0 / n : 0.0
        float pf = gl > 0 ? gp / gl : (gp > 0 ? 99.9 : 0.0)
        string txt = n > 0 ? "N " + str.tostring(n) + " | W " + str.tostring(math.round(winPct)) + "% | A " + p14Fmt(avgR) + " | T " + p14Fmt(totalR) + " | PF " + str.tostring(math.round(pf * 10.0) / 10.0) : "N 0 | W — | A — | T 0 | PF —"
        table.cell(tbl, m + 1, row, txt, text_color = totalR >= 0 ? #089981 : #f23645, text_size = size.tiny)
